   215→  // Export CSV
   216→  async function handleExport() {
   217→    setExporting(true)
   218→    try {
   219→      const params = new URLSearchParams({ reportType })
   220→      if (dateFrom) params.set('dateFrom', dateFrom)
   221→      if (dateTo) params.set('dateTo', dateTo)
   222→      if (categoryId) params.set('categoryId', categoryId)
   223→      if (departmentId) params.set('departmentId', departmentId)
   224→      if (projectId) params.set('projectId', projectId)
   225→      if (status) params.set('status', status)
   226→      if (productId) params.set('productId', productId)
   227→      if (supplierId) params.set('supplierId', supplierId)
   228→
   229→      const res = await fetch(`/api/reports/export/excel?${params}`)
   230→      if (!res.ok) {
   231→        toast.error('Export failed')
   232→        return
   233→      }
   234→      const blob = await res.blob()
   235→      const url = URL.createObjectURL(blob)
   236→      const a = document.createElement('a')
   237→      a.href = url
   238→      const disposition = res.headers.get('Content-Disposition')
   239→      const filename = disposition
   240→        ? disposition.split('filename=')[1]?.replace(/"/g, '')
   241→        : `${reportType}.csv`
   242→      a.download = filename
   243→      document.body.appendChild(a)
   244→      a.click()
   245→      document.body.removeChild(a)
   246→      URL.revokeObjectURL(url)
   247→      toast.success('Report exported successfully')
   248→    } catch {
   249→      toast.error('Export failed')
   250→    } finally {
   251→      setExporting(false)
   252→    }
   253→  }
