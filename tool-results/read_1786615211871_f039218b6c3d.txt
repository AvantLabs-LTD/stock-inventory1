   112→// ─── Helpers ─────────────────────────────────────────────────────────────

   114→function statusForGroup(group: ItemGroup): {
   115→  label: string
   116→  variant: 'destructive' | 'secondary' | 'outline'
   117→  className: string
   118→} {
   119→  if (group.totalInventory === 0) {
   120→    return { label: 'Out of Stock', variant: 'destructive', className: 'bg-red-100 text-red-700 border-red-200' }
   121→  }
   122→  const hasLow = group.specs.some((s) => s.minimumStock > 0 && s.quantity > 0 && s.quantity <= s.minimumStock)
   123→  const hasReserved = group.totalReserved > 0
   124→  if (hasLow) {
   125→    return { label: 'Low Stock', variant: 'secondary', className: 'bg-amber-100 text-amber-700 border-amber-200' }
   126→  }
   127→  if (hasReserved) {
   128→    return { label: 'Reserved', variant: 'secondary', className: 'bg-blue-100 text-blue-700 border-blue-200' }
   129→  }
   130→  return { label: 'In Stock', variant: 'secondary', className: 'bg-emerald-100 text-emerald-700 border-emerald-200' }
   131→}
   132→
   133→function statusForSpec(spec: SpecData): {
   134→  label: string
   135→  className: string
   136→} {
   137→  if (spec.quantity === 0) return { label: 'Out', className: 'bg-red-100 text-red-700' }
   138→  if (spec.minimumStock > 0 && spec.quantity > 0 && spec.quantity <= spec.minimumStock) {
   139→    return { label: 'Low', className: 'bg-amber-100 text-amber-700' }
   140→  }
   141→  if (spec.reservedQty > 0) return { label: 'Reserved', className: 'bg-blue-100 text-blue-700' }
   142→  return { label: 'OK', className: 'bg-emerald-100 text-emerald-700' }
   143→}
   144→
   145→function fmt(n: number): string {
   146→  return n.toLocaleString()
   147→}
   148→
   149→function formatDateTime(iso: string | null): string {
   150→  if (!iso) return '—'
   151→  const d = new Date(iso)
   152→  return d.toLocaleDateString('en-PK', { day: '2-digit', month: 'short' }) +
   153→    ' ' + d.toLocaleTimeString('en-PK', { hour: '2-digit', minute: '2-digit' })
   154→}

   158→function SummaryCards({ summary, loading }: { summary: SummaryStats | null; loading: boolean }) {
   159→  if (loading || !summary) {
   160→    return (
   161→      <div className="grid grid-cols-2 sm:grid-cols-4 lg:grid-cols-7 gap-3">
   162→        {Array.from({ length: 7 }).map((_, i) => (
   163→          <Skeleton key={i} className="h-20 rounded-lg" />
   164→        ))}
   165→      </div>
   166→    )
   167→  }
   168→
   169→  const cards = [
   170→    { label: 'Item Groups', value: summary.totalItems, icon: Boxes, color: 'text-slate-600', bg: 'bg-slate-50' },
   171→    { label: 'Specifications', value: summary.totalSpecs, icon: Package, color: 'text-violet-600', bg: 'bg-violet-50' },
   172→    { label: 'Total Inventory', value: summary.totalInventory, icon: TrendingUp, color: 'text-emerald-600', bg: 'bg-emerald-50' },
   173→    { label: 'Available', value: summary.totalAvailable, icon: Package, color: 'text-teal-600', bg: 'bg-teal-50' },
   174→    { label: 'Low Stock', value: summary.lowStockCount, icon: AlertTriangle, color: 'text-amber-600', bg: 'bg-amber-50' },
   175→    { label: 'Out of Stock', value: summary.outOfStockCount, icon: ArchiveX, color: 'text-red-600', bg: 'bg-red-50' },
   176→    { label: 'Reserved', value: summary.reservedCount, icon: Lock, color: 'text-blue-600', bg: 'bg-blue-50' },
   177→  ]
   178→
   179→  return (
   180→    <div className="grid grid-cols-2 sm:grid-cols-4 lg:grid-cols-7 gap-3">
   181→      {cards.map((c) => (
   182→        <Card key={c.label} className={`${c.bg} border-none shadow-sm`}>
   183→          <CardContent className="p-3">
   184→            <div className="flex items-center justify-between mb-1">
   185→              <span className="text-xs font-medium text-muted-foreground truncate">{c.label}</span>
   186→              <c.icon className={`h-3.5 w-3.5 ${c.color} opacity-60`} />
   187→            </div>
   188→            <p className={`text-lg font-bold ${c.color}`}>{fmt(c.value)}</p>
   189→          </CardContent>
   190→        </Card>
   191→      ))}
   192→    </div>
   193→  )
   194→}

   215→  const [pageSize, setPageSize] = useState('50')
   216→
   217→  // Dropdown options

   220→

   223→

   227→

   249→      return

   251→    setLoading(true)

   260→      if (stockFilter && stockFilter !== 'all') params.set('stockFilter', stockFilter)
   261→
   262→      const res = await fetch(`/api/inventory-items?${params.toString()}`)
   263→      if (res.ok) {
   264→        const json = await res.json()
   265→        setData(json.data || [])
   266→        setPagination(json.pagination || { page: 1, limit, total: 0, totalPages: 0 })
   267→        setItemNames(json.itemNames || [])
   268→        setWarehouses(json.warehouses || [])
   269→        setSummary(json.summary || null)
   270→      } else {
   271→        toast.error('Failed to load inventory data')
   272→      }
   273→    } catch {
   274→      toast.error('Network error loading inventory')
   275→    } finally {
   276→      setLoading(false)
   277→    }
   278→  }, [pagination.page, pageSize, search, itemNameFilter, warehouseFilter, stockFilter, canView])

   283→
   284→  // Reset page when filters change
   285→  useEffect(() => {
   286→    setPagination((p) => ({ ...p, page: 1 }))
   287→  }, [search, itemNameFilter, warehouseFilter, stockFilter, pageSize])
   288→

   292→    setExpandedItems((prev) => {

   294→      if (next.has(itemName)) {
   295→        next.delete(itemName)
   296→      } else {
   297→        next.add(itemName)
   298→      }
   299→      return next
   300→    })
   301→  }
   302→
   303→  const expandAll = () => {
   304→    setExpandedItems(new Set(data.map((g) => g.itemName)))

   322→      } else {

   326→      toast.error('Network error')

   328→      setDeleting(false)
   329→    }
   330→  }

   335→      const params = new URLSearchParams()

   337→      if (itemNameFilter && itemNameFilter !== 'all') params.set('itemName', itemNameFilter)

   339→      if (stockFilter && stockFilter !== 'all') params.set('stockFilter', stockFilter)
   340→
   341→      const res = await fetch(`/api/inventory-items/export?${params.toString()}`)
   342→      if (res.ok) {

   345→        const a = document.createElement('a')
   346→        a.href = url

   348→        a.click()
   349→        URL.revokeObjectURL(url)
   350→        toast.success('Export completed')
   351→      } else {
   352→        toast.error('Export failed')
   353→      }
   354→    } catch {
   355→      toast.error('Export error')
   356→    } finally {
   357→      setExporting(false)
   358→    }
   359→  }

   378→          })
   379→        }
   380→        fetchData()
   381→      } else {
   382→        const err = await res.json().catch(() => ({}))
   383→        toast.error(err.error || 'Import failed')
   384→      }
   385→    } catch {
   386→      toast.error('Import error')
   387→    } finally {

   389→      e.target.value = ''
   390→    }
