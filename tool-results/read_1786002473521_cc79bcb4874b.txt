   125→
   126→export function AddOpeningStockModal({
   127→  open,
   128→  onOpenChange,
   129→  onSaved,
   130→  projectFields,
   131→  editEntry,
   132→}: AddOpeningStockModalProps) {
   133→  const isEdit = !!editEntry
   134→
   135→  // ─── Product Search ────────────────────────────────────────────────────
   136→  const [productSearch, setProductSearch] = useState('')
   137→  const [productResults, setProductResults] = useState<ProductItem[]>([])
   138→  const [productSearchLoading, setProductSearchLoading] = useState(false)
   139→  const [selectedProduct, setSelectedProduct] = useState<ProductItem | null>(null)
   140→  const [showProductCreate, setShowProductCreate] = useState(false)
   141→
   142→  // ─── Form Fields ───────────────────────────────────────────────────────
   143→  const [warehouse, setWarehouse] = useState('')
   144→  const [storageLocation, setStorageLocation] = useState('')
   145→  const [openingQty, setOpeningQty] = useState<number>(0)
   146→  const [unitCost, setUnitCost] = useState<number>(0)
   147→  const [batchNumber, setBatchNumber] = useState('')
   148→  const [serialNumber, setSerialNumber] = useState('')
   149→  const [expiryDate, setExpiryDate] = useState('')
   150→  const [purchaseRef, setPurchaseRef] = useState('')
   151→  const [invoiceNumber, setInvoiceNumber] = useState('')
   152→  const [receivedDate, setReceivedDate] = useState(format(new Date(), 'yyyy-MM-dd'))
   153→  const [openingDate, setOpeningDate] = useState(format(new Date(), 'yyyy-MM-dd'))
   154→  const [remarks, setRemarks] = useState('')
   155→  const [internalNotes, setInternalNotes] = useState('')
   156→  const [projectQtys, setProjectQtys] = useState<ProjectQtyRow[]>([])
   157→
   158→  // ─── Quick Create Product ─────────────────────────────────────────────
   159→  const [newProductName, setNewProductName] = useState('')
   160→  const [newProductCategory, setNewProductCategory] = useState('')
   161→  const [newProductUnit, setNewProductUnit] = useState('pcs')
   162→  const [newProductVariantName, setNewProductVariantName] = useState('')
   163→  const [newProductParentId, setNewProductParentId] = useState('')
   164→  const [creatingProduct, setCreatingProduct] = useState(false)
   165→
   166→  // ─── Categories ──────────────────────────────────────────────────────────
   167→  const [categories, setCategories] = useState<CategoryItem[]>([])
   168→
   169→  // ─── Section Collapse ──────────────────────────────────────────────────
   170→  const [sectionsOpen, setSectionsOpen] = useState({
   171→    product: true,
   172→    inventory: true,
   173→    projects: false,
   174→    additional: false,
   175→  })
   176→
   177→  // ─── Submit ─────────────────────────────────────────────────────────────
   178→  const [submitting, setSubmitting] = useState(false)
   179→
   180→  // ─── Computed values ──────────────────────────────────────────────────
   181→  const inventoryValue = openingQty * unitCost
   182→
   183→  // ─── Initialize from edit entry ────────────────────────────────────────
   184→  useEffect(() => {
   185→    if (editEntry && open) {
   186→      // Set edit-only fields
   187→      setBatchNumber(editEntry.batchNumber || '')
   188→      setSerialNumber(editEntry.serialNumber || '')
   189→      setExpiryDate(editEntry.expiryDate ? format(new Date(editEntry.expiryDate), 'yyyy-MM-dd') : '')
   190→      setStorageLocation(editEntry.storageLocation || '')
   191→      setRemarks(editEntry.remarks || '')
   192→      setInternalNotes(editEntry.internalNotes || '')
   193→      setOpeningDate(editEntry.openingDate ? format(new Date(editEntry.openingDate), 'yyyy-MM-dd') : format(new Date(), 'yyyy-MM-dd'))
   194→      setWarehouse(editEntry.warehouse || '')
   195→      setOpeningQty(editEntry.quantity)
   196→      setUnitCost(editEntry.unitCost)
   197→
   198→      // Set project quantities
   199→      const pqRows = projectFields.map((pf) => {
   200→        const existing = editEntry.projectQtys?.find((pq) => pq.projectFieldId === pf.id)
   201→        return {
   202→          projectFieldId: pf.id,
   203→          requiredQty: existing?.requiredQty || 0,
   204→          batchQty: existing?.batchQty || 0,
   205→          reservedQty: existing?.reservedQty || 0,
   206→          consumedQty: existing?.consumedQty || 0,
   207→          orderedQty: existing?.orderedQty || 0,
   208→          toBeUsed: existing?.toBeUsed || 0,
   209→          remarks: existing?.remarks || '',
   210→        }
   211→      })
   212→      setProjectQtys(pqRows)
   213→
   214→      // Search and select the product (best effort)
   215→      setProductSearch('')
   216→      setSelectedProduct(null)
   217→      // We need to fetch the product info
   218→      fetch(`/api/products/${editEntry.productId}`)
   219→        .then((r) => {
   220→          if (r.ok) return r.json()
   221→          return null
   222→        })
   223→        .then((data) => {
   224→          if (data) {
   225→            setSelectedProduct({
   226→              id: data.id,
   227→              name: data.name,
   228→              code: data.code,
   229→              sku: data.sku,
   230→              unit: data.unit,
   231→              status: data.status,
   232→              categoryId: data.categoryId,
   233→              brand: data.brand,
   234→              parentProductId: data.parentProductId,
   235→              variantName: data.variantName,
   236→              category: data.category,
   237→              parent: data.parent,
   238→            })
   239→          }
   240→        })
   241→        .catch(() => {})
   242→    }
   243→  }, [editEntry, open, projectFields])
   244→
   245→  // ─── Reset on open for create mode ────────────────────────────────────
   246→  useEffect(() => {
   247→    if (open && !editEntry) {
   248→      setProductSearch('')
   249→      setProductResults([])
   250→      setSelectedProduct(null)
   251→      setShowProductCreate(false)
   252→      setWarehouse('')
   253→      setStorageLocation('')
   254→      setOpeningQty(0)
   255→      setUnitCost(0)
   256→      setBatchNumber('')
   257→      setSerialNumber('')
   258→      setExpiryDate('')
   259→      setPurchaseRef('')
   260→      setInvoiceNumber('')
   261→      setReceivedDate(format(new Date(), 'yyyy-MM-dd'))
   262→      setOpeningDate(format(new Date(), 'yyyy-MM-dd'))
   263→      setRemarks('')
   264→      setInternalNotes('')
   265→      setNewProductName('')
   266→      setNewProductCategory('')
   267→      setNewProductUnit('pcs')
   268→      setNewProductVariantName('')
   269→      setNewProductParentId('')
   270→      setProjectQtys(
   271→        projectFields.map((pf) => ({
   272→          projectFieldId: pf.id,
   273→          requiredQty: 0,
   274→          batchQty: 0,
   275→          reservedQty: 0,
   276→          consumedQty: 0,
   277→          orderedQty: 0,
   278→          toBeUsed: 0,
   279→          remarks: '',
   280→        }))
   281→      )
   282→      setSectionsOpen({ product: true, inventory: true, projects: false, additional: false })
   283→    }
   284→  }, [open, editEntry, projectFields])
   285→
   286→  // ─── Fetch categories ──────────────────────────────────────────────────
   287→  useEffect(() => {
   288→    if (!open) return
   289→    const fetchRefs = async () => {
   290→      try {
   291→        const cRes = await fetch('/api/categories?limit=500')
   292→        if (cRes.ok) {
   293→          const cJson = await cRes.json()
   294→          setCategories(cJson.data || [])
   295→        }
   296→      } catch {
   297→        // ignore
   298→      }
   299→    }
   300→    fetchRefs()
   301→  }, [open])
   302→
   303→  // ─── Product Search ────────────────────────────────────────────────────
   304→  const handleProductSearch = useCallback(async (query: string) => {
   305→    setProductSearch(query)
   306→    setSelectedProduct(null)
   307→    if (!query || query.length < 2) {
   308→      setProductResults([])
   309→      return
   310→    }
   311→    setProductSearchLoading(true)
   312→    try {
   313→      const res = await fetch(`/api/products?search=${encodeURIComponent(query)}&limit=20&status=ACTIVE`)
   314→      if (!res.ok) throw new Error('Failed')
   315→      const json = await res.json()
   316→      setProductResults(json.data || [])
   317→    } catch {
   318→      setProductResults([])
   319→    } finally {
   320→      setProductSearchLoading(false)
   321→    }
   322→  }, [])
   323→
   324→  const handleSelectProduct = useCallback((product: ProductItem) => {
   325→    setSelectedProduct(product)
   326→    setProductSearch(product.name)
   327→    setProductResults([])
   328→    setShowProductCreate(false)
   329→  }, [editEntry])
   330→
   331→  // ─── Quick Create Product ──────────────────────────────────────────────
   332→  const handleQuickCreate = useCallback(async () => {
   333→    if (!newProductName.trim()) return
   334→    setCreatingProduct(true)
   335→    try {
   336→      const body: Record<string, unknown> = {
   337→        name: newProductName.trim(),
   338→        unit: newProductUnit,
   339→        categoryId: newProductCategory || undefined,
   340→        variantName: newProductVariantName || undefined,
   341→        parentProductId: newProductParentId || undefined,
   342→      }
   343→      const res = await fetch('/api/opening-stock/products/quick-create', {
   344→        method: 'POST',
   345→        headers: { 'Content-Type': 'application/json' },
   346→        body: JSON.stringify(body),
   347→      })
   348→      if (!res.ok) {
   349→        const err = await res.json()
   350→        throw new Error(err.error || 'Failed to create product')
   351→      }
   352→      const product = await res.json()
   353→      toast.success(`Product "${product.name}" created`)
   354→      handleSelectProduct({
   355→        id: product.id,
   356→        name: product.name,
   357→        code: product.code,
   358→        sku: product.sku,
   359→        unit: product.unit,
   360→        status: 'ACTIVE',
   361→        categoryId: product.categoryId,
   362→        brand: null,
   363→        parentProductId: product.parentProductId,
   364→        variantName: product.variantName,
   365→        category: null,
   366→        parent: null,
   367→      })
   368→      setShowProductCreate(false)
   369→      setNewProductName('')
   370→    } catch (err) {
   371→      toast.error(err instanceof Error ? err.message : 'Failed to create product')
   372→    } finally {
   373→      setCreatingProduct(false)
   374→    }
   375→  }, [newProductName, newProductUnit, newProductCategory, newProductVariantName, newProductParentId, handleSelectProduct])
   376→
   377→  // ─── Project Qty Update ────────────────────────────────────────────────
   378→  const updateProjectQty = useCallback((idx: number, field: keyof ProjectQtyRow, value: number | string) => {
   379→    setProjectQtys((prev) => {
   380→      const next = [...prev]
   381→      next[idx] = { ...next[idx], [field]: value }
   382→      return next
   383→    })
   384→  }, [])
   385→
   386→  // ─── Toggle Section ───────────────────────────────────────────────────
   387→  const toggleSection = useCallback((key: keyof typeof sectionsOpen) => {
   388→    setSectionsOpen((prev) => ({ ...prev, [key]: !prev[key] }))
   389→  }, [sectionsOpen])
   390→
   391→  // ─── Submit ────────────────────────────────────────────────────────────
   392→  const handleSubmit = useCallback(async () => {
   393→    if (!selectedProduct && !editEntry) {
   394→      toast.error('Please select a product')
   395→      return
   396→    }
   397→    if (!isEdit && openingQty <= 0) {
   398→      toast.error('Opening quantity must be greater than 0')
   399→      return
   400→    }
   401→
   402→    setSubmitting(true)
   403→    try {
   404→      if (isEdit && editEntry) {
   405→        // Edit mode - only limited fields
   406→        const updateData: Record<string, unknown> = {
   407→          remarks,
   408→          internalNotes,
   409→          batchNumber: batchNumber || null,
   410→          serialNumber: serialNumber || null,
   411→          expiryDate: expiryDate || null,
   412→          storageLocation: storageLocation || null,
   413→        }
   414→
   415→        const res = await fetch(`/api/opening-stock/${editEntry.id}`, {
   416→          method: 'PUT',
   417→          headers: { 'Content-Type': 'application/json' },
   418→          body: JSON.stringify(updateData),
   419→        })
   420→        if (!res.ok) {
   421→          const err = await res.json()
   422→          throw new Error(err.error || 'Failed to update')
   423→        }
   424→        toast.success('Opening stock updated')
   425→      } else {
   426→        // Create mode
   427→        const body: Record<string, unknown> = {
   428→          productId: selectedProduct!.id,
   429→          quantity: openingQty,
   430→          unitCost,
   431→          warehouse: warehouse || null,
   432→          storageLocation: storageLocation || null,
   433→          batchNumber: batchNumber || null,
   434→          serialNumber: serialNumber || null,
   435→          expiryDate: expiryDate || null,
   436→          purchaseReference: purchaseRef || null,
   437→          invoiceNumber: invoiceNumber || null,
   438→          receivedDate: receivedDate || null,
   439→          openingDate: openingDate || null,
   440→          remarks: remarks || null,
   441→          internalNotes: internalNotes || null,
   442→          projectQtys: projectQtys
   443→            .filter((pq) => pq.requiredQty > 0)
   444→            .map((pq) => ({
   445→              projectFieldId: pq.projectFieldId,
   446→              requiredQty: pq.requiredQty,
   447→              batchQty: pq.batchQty,
   448→              reservedQty: pq.reservedQty,
   449→              consumedQty: pq.consumedQty,
   450→              orderedQty: pq.orderedQty,
   451→              toBeUsed: pq.toBeUsed,
   452→              remarks: pq.remarks || null,
   453→            })),
   454→        }
   455→
   456→        const res = await fetch('/api/opening-stock', {
   457→          method: 'POST',
   458→          headers: { 'Content-Type': 'application/json' },
   459→          body: JSON.stringify(body),
   460→        })
   461→        if (!res.ok) {
   462→          const err = await res.json()
   463→          throw new Error(err.error || 'Failed to create')
   464→        }
   465→        toast.success('Opening stock created successfully')
   466→      }
   467→
   468→      onOpenChange(false)
   469→      onSaved()
   470→    } catch (err) {
   471→      toast.error(err instanceof Error ? err.message : 'Operation failed')
   472→    } finally {
   473→      setSubmitting(false)
   474→    }
   475→  }, [
   476→    selectedProduct,
   477→    editEntry,
   478→    isEdit,
   479→    openingQty,
   480→    unitCost,
   481→    warehouse,
   482→    storageLocation,
   483→    batchNumber,
   484→    serialNumber,
   485→    expiryDate,
   486→    purchaseRef,
   487→    invoiceNumber,
   488→    receivedDate,
   489→    openingDate,
   490→    remarks,
   491→    internalNotes,
   492→    projectQtys,
   493→    onOpenChange,
   494→    onSaved,
   495→  ])
   496→
   497→  // ─── Section Component ──────────────────────────────────────────────────
   498→  const SectionHeader = ({
   499→    sectionKey,
   500→    title,
   501→    subtitle,
   502→    badge,
   503→  }: {
   504→    sectionKey: keyof typeof sectionsOpen
   505→    title: string
   506→    subtitle?: string
   507→    badge?: string
   508→  }) => (
   509→    <CollapsibleTrigger className="flex w-full items-center justify-between rounded-lg border bg-muted/30 px-4 py-3 hover:bg-muted/50 transition-colors">
   510→      <div className="flex items-center gap-3">
   511→        <h3 className="text-sm font-semibold">{title}</h3>
   512→        {subtitle && <span className="text-xs text-muted-foreground">{subtitle}</span>}
   513→        {badge && (
   514→          <Badge variant="secondary" className="text-xs">
   515→            {badge}
   516→          </Badge>
   517→        )}
   518→      </div>
   519→      {sectionsOpen[sectionKey] ? (
   520→        <ChevronUp className="h-4 w-4 text-muted-foreground" />
   521→      ) : (
   522→        <ChevronDown className="h-4 w-4 text-muted-foreground" />
   523→      )}
   524→    </CollapsibleTrigger>
   525→  )
   526→
   527→  return (
   528→    <Dialog open={open} onOpenChange={onOpenChange}>
   529→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-hidden flex flex-col">
   530→        <DialogHeader>
   531→          <DialogTitle>{isEdit ? 'Edit Opening Stock' : 'Add Opening Stock'}</DialogTitle>
   532→          <DialogDescription>
   533→            {isEdit ? 'Update inventory metadata and project quantities.' : 'Create a new opening stock entry for inventory.'}
   534→          </DialogDescription>
   535→        </DialogHeader>
   536→
   537→        <div className="flex-1 overflow-y-auto space-y-3 pr-1">
   538→          {/* ─── 1. Product Information ─────────────────────────────────── */}
   539→          <Collapsible open={sectionsOpen.product} onOpenChange={() => toggleSection('product')}>
   540→            <SectionHeader
   541→              sectionKey="product"
   542→              title="Product Information"
   543→              subtitle="Search and select the product"
   544→              badge={selectedProduct ? 'Selected' : undefined}
   545→            />
   546→            <CollapsibleContent className="space-y-4 pt-3 px-1">
   547→              {!showProductCreate ? (
   548→                <>
   549→                  <div className="relative">
   550→                    <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   551→                    <Input
   552→                      placeholder="Search by product name, SKU, or code..."
   553→                      value={isEdit && selectedProduct ? selectedProduct.name : productSearch}
   554→                      onChange={(e) => {
   555→                        if (isEdit) return // no change in edit mode
   556→                        handleProductSearch(e.target.value)
   557→                      }}
   558→                      className="pl-9"
   559→                      disabled={isEdit}
   560→                    />
   561→                    {productSearchLoading && (
   562→                      <Loader2 className="absolute right-3 top-1/2 h-4 w-4 -translate-y-1/2 animate-spin text-muted-foreground" />
   563→                    )}
   564→                  </div>
   565→
   566→                  {/* Search Results Dropdown */}
   567→                  {productResults.length > 0 && !selectedProduct && (
   568→                    <div className="rounded-md border bg-background shadow-lg max-h-48 overflow-y-auto">
   569→                      {productResults.map((p) => (
   570→                        <button
   571→                          key={p.id}
   572→                          className="flex w-full items-center justify-between px-3 py-2 text-sm hover:bg-muted/50 transition-colors"
   573→                          onClick={() => handleSelectProduct(p)}
   574→                        >
   575→                          <div>
   576→                            <p className="font-medium">{p.name}</p>
   577→                            <p className="text-xs text-muted-foreground">
   578→                              {p.code} | {p.sku} | {p.category?.name || 'No Category'}
   579→                              {p.parentProductId && ` (Variant: ${p.variantName || 'N/A'})`}
   580→                            </p>
   581→                          </div>
   582→                          <Badge variant="outline" className="text-xs">
   583→                            {p.unit}
   584→                          </Badge>
   585→                        </button>
   586→                      ))}
   587→                    </div>
   588→                  )}
   589→
   590→                  {/* Product not found + create prompt */}
   591→                  {productSearch.length >= 2 && !productSearchLoading && productResults.length === 0 && !selectedProduct && (
   592→                    <div className="flex items-center gap-2 rounded-md border border-dashed p-3">
   593→                      <AlertCircle className="h-4 w-4 text-amber-500" />
   594→                      <span className="text-sm text-muted-foreground">
   595→                        Product not found.{' '}
   596→                        <button
   597→                          className="text-primary font-medium hover:underline"
   598→                          onClick={() => setShowProductCreate(true)}
   599→                        >
   600→                          Create Product?
   601→                        </button>

   660→                        value={newProductName}

   757→                    min={0}
   758→                    step={0.01}
