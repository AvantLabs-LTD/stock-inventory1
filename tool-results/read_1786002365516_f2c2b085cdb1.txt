   317→function StockStatusBadge({ status }: { status: string }) {
   318→  switch (status) {
   319→    case 'HEALTHY':
   320→      return (
   321→        <Badge variant="outline" className="border-emerald-500/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-400">
   322→          Healthy
   323→        </Badge>
   324→      )
   325→    case 'LOW':
   326→      return (
   327→        <Badge variant="outline" className="border-amber-500/30 bg-amber-500/10 text-amber-700 dark:text-amber-400">
   328→          Low Stock
   329→        </Badge>
   330→      )
   331→    case 'CRITICAL':
   332→      return (
   333→        <Badge variant="outline" className="border-red-500/30 bg-red-500/10 text-red-700 dark:text-red-400">
   334→          Critical
   335→        </Badge>
   336→      )
   337→    case 'OUT_OF_STOCK':
   338→      return (
   339→        <Badge variant="outline" className="border-gray-500/30 bg-gray-500/10 text-gray-700 dark:text-gray-400">
   340→          Out of Stock
   341→        </Badge>
   342→      )
   343→    default:
   344→      return <Badge variant="secondary">{status}</Badge>
   345→  }
   346→}

   359→export function OpeningStockPageV2() {
   360→  const user = useAuthStore((s) => s.user)
   361→  const canManage = user ? hasPermission(user.role, 'stock', 'manage') : false
   362→  const canView = user ? hasPermission(user.role, 'stock', 'view') : false
   363→
   364→  // ─── Data State ──────────────────────────────────────────────────────────
   365→  const [entries, setEntries] = useState<OpeningStockEntry[]>([])
   366→  const [summary, setSummary] = useState<SummaryData | null>(null)
   367→  const [categories, setCategories] = useState<CategoryItem[]>([])
   368→  const [projectFields, setProjectFields] = useState<ProjectField[]>([])
   369→  const [total, setTotal] = useState(0)
   370→  const [loading, setLoading] = useState(true)
   371→  const [summaryLoading, setSummaryLoading] = useState(true)
   372→
   373→  // ─── Filter State ───────────────────────────────────────────────────────
   374→  const [search, setSearch] = useState('')
   375→  const [categoryId, setCategoryId] = useState('')
   376→  const [warehouse, setWarehouse] = useState('')
   377→  const [status, setStatus] = useState('ACTIVE')
   378→  const [stockStatus, setStockStatus] = useState('')
   379→  const [projectId, setProjectId] = useState('')
   380→  const [sortBy, setSortBy] = useState('openingDate')
   381→  const [sortOrder, setSortOrder] = useState<'asc' | 'desc'>('desc')
   382→  const [page, setPage] = useState(1)
   383→
   384→  // ─── Column Visibility ─────────────────────────────────────────────────
   385→  const [visibleColumns, setVisibleColumns] = useState<Set<string>>(
   386→    () => new Set(DEFAULT_COLUMNS.filter((c) => c.defaultVisible).map((c) => c.key))
   387→  )
   388→
   389→  // ─── Selection ───────────────────────────────────────────────────────────
   390→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   391→
   392→  // ─── Dialogs ────────────────────────────────────────────────────────────
   393→  const [addModalOpen, setAddModalOpen] = useState(false)
   394→  const [importDialogOpen, setImportDialogOpen] = useState(false)
   395→  const [projectConfigOpen, setProjectConfigOpen] = useState(false)
   396→  const [detailDialogOpen, setDetailDialogOpen] = useState(false)
   397→  const [detailEntry, setDetailEntry] = useState<OpeningStockEntry | null>(null)
   398→  const [archiveDialogOpen, setArchiveDialogOpen] = useState(false)
   399→  const [archiveId, setArchiveId] = useState<string | null>(null)
   400→  const [restoreId, setRestoreId] = useState<string | null>(null)
   401→  const [restoreDialogOpen, setRestoreDialogOpen] = useState(false)
   402→
   403→  // ─── Ledger ─────────────────────────────────────────────────────────────
   404→  const [ledgerDrawerOpen, setLedgerDrawerOpen] = useState(false)
   405→  const [ledgerEntryId, setLedgerEntryId] = useState<string | null>(null)
   406→
   407→  const searchTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   408→
   409→  // ─── Count active filters ───────────────────────────────────────────────
   410→  const activeFilterCount = useMemo(() => {
   411→    let count = 0
   412→    if (search) count++
   413→    if (categoryId) count++
   414→    if (warehouse) count++
   415→    if (status !== 'ACTIVE') count++
   416→    if (stockStatus) count++
   417→    if (projectId) count++
   418→    return count
   419→  }, [search, categoryId, warehouse, status, stockStatus, projectId])
   420→
   421→  const resetFilters = useCallback(() => {
   422→    setSearch('')
   423→    setCategoryId('')
   424→    setWarehouse('')
   425→    setStatus('ACTIVE')
   426→    setStockStatus('')
   427→    setProjectId('')
   428→    setPage(1)
   429→  }, [])
   430→
   431→  // ─── Fetch Data ─────────────────────────────────────────────────────────
   432→  const fetchEntries = useCallback(async () => {
   433→    if (!canView) return
   434→    setLoading(true)
   435→    try {
   436→      const params = new URLSearchParams({
   437→        page: String(page),
   438→        limit: String(PAGE_SIZE),
   439→        sortBy,
   440→        sortOrder,
   441→      })
   442→      if (search) params.set('search', search)
   443→      if (categoryId) params.set('categoryId', categoryId)
   444→      if (warehouse) params.set('warehouse', warehouse)
   445→      if (status) params.set('status', status)
   446→      if (stockStatus) params.set('stockStatus', stockStatus)
   447→
   448→      const res = await fetch(`/api/opening-stock?${params}`)
   449→      if (!res.ok) throw new Error('Failed to fetch')
   450→      const json = await res.json()
   451→      setEntries(json.data || [])
   452→      setTotal(json.pagination?.total || 0)
   453→
   454→      // Update project fields from response
   455→      if (json.projectFields) {
   456→        setProjectFields(json.projectFields)
   457→      }
   458→    } catch (err) {
   459→      console.error('Fetch error:', err)
   460→      toast.error('Failed to load inventory data')
   461→    } finally {
   462→      setLoading(false)
   463→    }
   464→  }, [page, search, categoryId, warehouse, status, stockStatus, sortBy, sortOrder, canView])
   465→
   466→  const fetchSummary = useCallback(async () => {
   467→    if (!canView) return
   468→    setSummaryLoading(true)
   469→    try {
   470→      const res = await fetch('/api/opening-stock/summary')
   471→      if (!res.ok) throw new Error('Failed to fetch summary')
   472→      const json = await res.json()
   473→      setSummary(json)
   474→    } catch (err) {
   475→      console.error('Summary error:', err)
   476→    } finally {
   477→      setSummaryLoading(false)
   478→    }
   479→  }, [canView])
   480→
   481→  const fetchCategories = useCallback(async () => {
   482→    try {
   483→      const res = await fetch('/api/categories?limit=500')
   484→      if (!res.ok) throw new Error('Failed to fetch categories')
   485→      const json = await res.json()
   486→      setCategories(json.data || [])
   487→    } catch {
   488→      // ignore
   489→    }
   490→  }, [])
   491→
   492→  const fetchProjectFields = useCallback(async () => {
   493→    try {
   494→      const res = await fetch('/api/opening-stock/projects')
   495→      if (!res.ok) throw new Error('Failed to fetch projects')
   496→      const json = await res.json()
   497→      setProjectFields(json.data || [])
   498→    } catch {
   499→      // ignore
   500→    }
   501→  }, [])
   502→
   503→  useEffect(() => {
   504→    fetchEntries()
   505→  }, [fetchEntries])
   506→
   507→  useEffect(() => {
   508→    fetchSummary()
   509→    fetchCategories()
   510→    fetchProjectFields()
   511→  }, [fetchSummary, fetchCategories, fetchProjectFields])
   512→
   513→  // ─── Debounced search ──────────────────────────────────────────────────
   514→  const handleSearchChange = useCallback((val: string) => {
   515→    setSearch(val)
   516→    setPage(1)
   517→    if (searchTimerRef.current) clearTimeout(searchTimerRef.current)
   518→    searchTimerRef.current = setTimeout(() => {
   519→      // fetchEntries will re-run via the search dependency change
   520→    }, 300)
   521→  }, [])
   522→
   523→  // ─── Sort Handler ───────────────────────────────────────────────────────
   524→  const handleSort = useCallback((field: string) => {
   525→    if (sortBy === field) {
   526→      setSortOrder((prev) => (prev === 'asc' ? 'desc' : 'asc'))
   527→    } else {
   528→      setSortBy(field)
   529→      setSortOrder('asc')
   530→    }
   531→    setPage(1)
   532→  }, [sortBy])
   533→
   534→  // ─── Column Toggle ──────────────────────────────────────────────────────
   535→  const toggleColumn = useCallback((key: string) => {
   536→    setVisibleColumns((prev) => {
   537→      const next = new Set(prev)
   538→      if (next.has(key)) {
   539→        next.delete(key)
   540→      } else {
   541→        next.add(key)
   542→      }
   543→      return next
   544→    })
   545→  }, [])
   546→
   547→  // ─── Select / Deselect All ─────────────────────────────────────────────
   548→  const allSelected = entries.length > 0 && entries.every((e) => selectedIds.has(e.id))
   549→  const handleSelectAll = useCallback(() => {
   550→    if (allSelected) {
   551→      setSelectedIds(new Set())
   552→    } else {
   553→      setSelectedIds(new Set(entries.map((e) => e.id)))
   554→    }
   555→  }, [allSelected, entries])
   556→
   557→  const handleSelectRow = useCallback((id: string) => {
   558→    setSelectedIds((prev) => {
   559→      const next = new Set(prev)
   560→      if (next.has(id)) next.delete(id)
   561→      else next.add(id)
   562→      return next
   563→    })
   564→  }, [])
   565→
   566→  // ─── Actions ────────────────────────────────────────────────────────────
   567→  const handleArchive = useCallback(async () => {
   568→    if (!archiveId) return
   569→    try {
   570→      const res = await fetch(`/api/opening-stock/${archiveId}`, { method: 'DELETE' })
   571→      if (!res.ok) {
   572→        const err = await res.json()
   573→        throw new Error(err.error || 'Failed to archive')
   574→      }
   575→      toast.success('Entry archived successfully')
   576→      fetchEntries()
   577→      fetchSummary()
   578→    } catch (err) {
   579→      toast.error(err instanceof Error ? err.message : 'Failed to archive')
   580→    } finally {
   581→      setArchiveDialogOpen(false)
   582→      setArchiveId(null)
   583→    }
   584→  }, [archiveId, fetchEntries, fetchSummary])
   585→
   586→  const handleRestore = useCallback(async () => {
   587→    if (!restoreId) return
   588→    try {
   589→      const res = await fetch(`/api/opening-stock/${restoreId}/restore`, { method: 'PATCH' })
   590→      if (!res.ok) {
   591→        const err = await res.json()
   592→        throw new Error(err.error || 'Failed to restore')
   593→      }
   594→      toast.success('Entry restored successfully')
   595→      fetchEntries()
   596→      fetchSummary()
   597→    } catch (err) {
   598→      toast.error(err instanceof Error ? err.message : 'Failed to restore')
   599→    } finally {
   600→      setRestoreDialogOpen(false)
   601→      setRestoreId(null)
   602→    }
   603→  }, [restoreId, fetchEntries, fetchSummary])
   604→
