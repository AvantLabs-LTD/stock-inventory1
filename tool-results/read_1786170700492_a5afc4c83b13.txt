   109→   109→function statusForGroup(group: ItemGroup): {
   110→   110→  label: string
   111→   111→  variant: 'destructive' | 'secondary' | 'outline'
   112→   112→  className: string
   113→   113→} {
   114→   114→  if (group.totalInventory === 0) {
   115→   115→    return { label: 'Out of Stock', variant: 'destructive', className: 'bg-red-100 text-red-700 border-red-200' }
   116→   116→  }
   117→   117→  const hasLow = group.specs.some((s) => s.minimumStock > 0 && s.quantity > 0 && s.quantity <= s.minimumStock)
   118→   118→  const hasReserved = group.totalReserved > 0
   119→   119→  if (hasLow) {
   120→   120→    return { label: 'Low Stock', variant: 'secondary', className: 'bg-amber-100 text-amber-700 border-amber-200' }
   121→   121→  }
   122→   122→  if (hasReserved) {
   123→   123→    return { label: 'Reserved', variant: 'secondary', className: 'bg-blue-100 text-blue-700 border-blue-200' }
   124→   124→  }
   125→   125→  return { label: 'In Stock', variant: 'secondary', className: 'bg-emerald-100 text-emerald-700 border-emerald-200' }
   126→   126→}
   127→   127→
   128→   128→function statusForSpec(spec: SpecData): {
   129→   129→  label: string
   130→   130→  className: string
   131→   131→} {
   132→   132→  if (spec.quantity === 0) return { label: 'Out', className: 'bg-red-100 text-red-700' }
   133→   133→  if (spec.minimumStock > 0 && spec.quantity > 0 && spec.quantity <= spec.minimumStock) {
   134→   134→    return { label: 'Low', className: 'bg-amber-100 text-amber-700' }
   135→   135→  }
   136→   136→  if (spec.reservedQty > 0) return { label: 'Reserved', className: 'bg-blue-100 text-blue-700' }
   137→   137→  return { label: 'OK', className: 'bg-emerald-100 text-emerald-700' }
   138→   138→}

   140→   140→function fmt(n: number): string {

   305→   305→    } catch {
   306→   306→      toast.error('Network error')
   307→   307→    } finally {
   308→   308→      setDeleting(false)

   310→   310→  }

   317→   317→      if (itemNameFilter && itemNameFilter !== 'all') params.set('itemName', itemNameFilter)
   318→   318→      if (warehouseFilter && warehouseFilter !== 'all') params.set('warehouse', warehouseFilter)
