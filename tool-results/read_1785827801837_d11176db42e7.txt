   322→  async function onSubmit(values: OpeningFormValues) {

   325→      const res = await fetch('/api/stock/opening', {
   326→        method: 'POST',

   328→        body: JSON.stringify(values),

   335→        const data = await res.json()

   337→      }

   339→      toast.error('Network error. Please try again.')
   340→    } finally {

   345→  // ─── Export Excel ──────────────────────────────────────────────────────
   346→
   347→  async function handleExportExcel() {
   348→    setExporting(true)
   349→    try {
   350→      const res = await fetch('/api/stock/opening/export')
   351→      if (!res.ok) {
   352→        const data = await res.json().catch(() => ({}))
   353→        toast.error(data.error || 'Export failed')
   354→        return
   355→      }
   356→      const blob = await res.blob()
   357→      const url = URL.createObjectURL(blob)
   358→      const a = document.createElement('a')
   359→      a.href = url
   360→      const cd = res.headers.get('Content-Disposition') || ''
   361→      const match = cd.match(/filename="?(.+?)"?$/)
   362→      a.download = match ? decodeURIComponent(match[1]) : `Opening Stock - Electronic Connectors_${new Date().toISOString().split('T')[0]}.xlsx`
   363→      document.body.appendChild(a)
   364→      a.click()
   365→      a.remove()
   366→      URL.revokeObjectURL(url)
   367→      toast.success('Excel file downloaded successfully')
   368→    } catch {
   369→      toast.error('Failed to export. Please try again.')
   370→    } finally {
   371→      setExporting(false)
   372→    }
   373→  }

   375→  // ─── Import Excel ─────────────────────────────────────────────────────

   377→  async function handleImportExcel() {
   378→    if (!importFile) {
   379→      toast.error('Please select a file first')

   382→    setImporting(true)
   383→    setImportResult(null)

   387→
   388→      const res = await fetch('/api/stock/opening/import', {
   389→        method: 'POST',
   390→        body: formData,
   391→      })
   392→      const data = await res.json()
   393→      if (res.ok) {
   394→        setImportResult(data)
   395→        toast.success(data.message || 'Import completed')
   396→        fetchEntries()
   397→      } else {
   398→        toast.error(data.error || 'Import failed')
   399→        setImportResult({ success: false, message: data.error || 'Import failed', imported: 0, updated: 0, skipped: 0, errors: [] })
   400→      }
   401→    } catch {
   402→      toast.error('Failed to import. Please try again.')
   403→    } finally {
   404→      setImporting(false)
   405→    }
