     4→import { format } from 'date-fns'
     5→import {

    31→} from '@/components/ui/dialog'

    60→// ─── Types ───────────────────────────────────────────────────────────────────
    61→
    62→interface StockInfo {
    63→  totalInStock: number
    64→  totalIssued: number
    65→  totalReserved: number
    66→  availableInStore: number
    67→}
    68→
    69→interface RequisitionItemData {
    70→  id: string
    71→  productId: string
    72→  specDescription: string | null
    73→  requiredQty: number
    74→  issuedQty: number
    75→  remarks: string | null
    76→  product: { id: string; name: string; code: string; sku: string; unit: string }
    77→  stockInfo?: StockInfo
    78→}
    79→
    80→interface RequisitionData {
    81→  id: string
    82→  requisitionNo: string
    83→  date: string
    84→  departmentId: string
    85→  projectId: string
    86→  requestedBy: string
    87→  employeeName: string
    88→  issuedByName: string | null
    89→  receivedByName: string | null
    90→  approvedBy: string | null
    91→  approvedAt: string | null
    92→  completedBy: string | null
    93→  completedAt: string | null
    94→  status: string
    95→  remarks: string | null
    96→  createdAt: string
    97→  updatedAt: string
    98→  department: { id: string; name: string; code: string }
    99→  project: { id: string; name: string; code: string }
   100→  requestedByUser: { id: string; name: string; email: string }
   101→  items: RequisitionItemData[]
   102→}
   103→
   104→// ─── Status Colors ──────────────────────────────────────────────────────────
   105→
   106→const statusColors: Record<string, string> = {
   107→  PENDING: 'bg-amber-100 text-amber-800 dark:bg-amber-900/30 dark:text-amber-400',
   108→  APPROVED: 'bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-400',
   109→  PARTIAL_APPROVED: 'bg-blue-100 text-blue-800 dark:bg-blue-900/30 dark:text-blue-400',
   110→  REJECTED: 'bg-red-100 text-red-800 dark:bg-red-900/30 dark:text-red-400',
   111→  CLOSED: 'bg-gray-200 text-gray-700 dark:bg-gray-800 dark:text-gray-300',
   112→  COMPLETED: 'bg-emerald-100 text-emerald-800 dark:bg-emerald-900/30 dark:text-emerald-400',
   113→  CANCELLED: 'bg-gray-100 text-gray-800 dark:bg-gray-900/30 dark:text-gray-400',
   114→}
   115→
   116→// ─── Props ──────────────────────────────────────────────────────────────────
   117→
   118→interface Props {
   119→  requisitionId: string | null
   120→  open: boolean
   121→  onOpenChange: (open: boolean) => void
   122→  onAction: () => void
   123→}
   124→
   125→// ─── Component ───────────────────────────────────────────────────────────────
   126→
   127→export function RequisitionDetailDialog({ requisitionId, open, onOpenChange, onAction }: Props) {
   128→  const user = useAuthStore((s) => s.user)
   129→  const canApprove = user ? hasPermission(user.role, 'inventory_requests', 'approve') : false
   130→  const canReject = user ? hasPermission(user.role, 'inventory_requests', 'reject') : false
   131→  const canEdit = user ? hasPermission(user.role, 'inventory_requests', 'edit') : false
   132→
   133→  const [data, setData] = useState<RequisitionData | null>(null)
   134→  const [loading, setLoading] = useState(false)
   135→  const [actionLoading, setActionLoading] = useState(false)
   136→
   137→  // Approve
   138→  const [approveOpen, setApproveOpen] = useState(false)
   139→  const [approveQtys, setApproveQtys] = useState<Record<string, number>>({})
   140→
   141→  // Reject
   142→  const [rejectOpen, setRejectOpen] = useState(false)
   143→  const [rejectRemarks, setRejectRemarks] = useState('')
   144→
   145→  // Complete
   146→  const [completeOpen, setCompleteOpen] = useState(false)
   147→
   148→  // Edit signature
   149→  const [editIssuedBy, setEditIssuedBy] = useState(false)
   150→  const [editReceivedBy, setEditReceivedBy] = useState(false)
   151→  const [issuedByName, setIssuedByName] = useState('')
   152→  const [receivedByName, setReceivedByName] = useState('')
   153→  const [savingSignatures, setSavingSignatures] = useState(false)
   154→
   155→  // ─── Fetch ──────────────────────────────────────────────────────────────
   156→  const fetchData = useCallback(async () => {
   157→    if (!requisitionId) { setData(null); return }
   158→    setLoading(true)
   159→    try {
   160→      const r = await fetch(`/api/requisitions/${requisitionId}`)
   161→      const d = await r.json()
   162→      setData(d.data || null)
   163→      if (d.data) {
   164→        setIssuedByName(d.data.issuedByName || '')
   165→        setReceivedByName(d.data.receivedByName || '')
   166→      }
   167→    } catch { setData(null) } finally { setLoading(false) }
   168→  }, [requisitionId])
   169→
   170→  useEffect(() => {
   171→    if (!open) return
   172→    setEditIssuedBy(false)
   173→    setEditReceivedBy(false)
   174→    fetchData()
   175→  }, [open, fetchData])
   176→
   177→  // ─── Approve ─────────────────────────────────────────────────────────────
   178→  async function handleApprove() {
   179→    if (!requisitionId || !data) return
   180→    setActionLoading(true)
   181→    try {
   182→      const itemsPayload = data.items.map((item) => ({
   183→        itemId: item.id,
   184→        approvedQty: approveQtys[item.id] ?? item.requiredQty,
   185→      }))
   186→      const res = await fetch(`/api/requisitions/${requisitionId}/approve`, {
   187→        method: 'PUT',
   188→        headers: { 'Content-Type': 'application/json' },
   189→        body: JSON.stringify({ items: itemsPayload }),
   190→      })
   191→      if (res.ok) {
   192→        toast.success('Requisition approved')
   193→        setApproveOpen(false)
   194→        onOpenChange(false)
   195→        onAction()
   196→      } else {
   197→        const d = await res.json()
   198→        toast.error(d.error || 'Failed to approve')
   199→      }
   200→    } catch { toast.error('Network error') } finally { setActionLoading(false) }
   201→  }
   202→
   203→  // ─── Reject ──────────────────────────────────────────────────────────────
   204→  async function handleReject() {
   205→    if (!requisitionId) return
   206→    setActionLoading(true)
   207→    try {
   208→      const res = await fetch(`/api/requisitions/${requisitionId}/reject`, {
   209→        method: 'PUT',
   210→        headers: { 'Content-Type': 'application/json' },
   211→        body: JSON.stringify({ remarks: rejectRemarks }),
   212→      })
   213→      if (res.ok) {
   214→        toast.success('Requisition rejected')
   215→        setRejectOpen(false)
   216→        onOpenChange(false)
   217→        onAction()
   218→      } else {
   219→        const d = await res.json()
   220→        toast.error(d.error || 'Failed to reject')
   221→      }
   222→    } catch { toast.error('Network error') } finally { setActionLoading(false) }
   223→  }
   224→
   225→  // ─── Complete (Issue & Close) ───────────────────────────────────────────
   226→  async function handleComplete() {
   227→    if (!requisitionId) return
   228→    setActionLoading(true)
   229→    try {
   230→      const res = await fetch(`/api/requisitions/${requisitionId}/complete`, { method: 'PUT' })
   231→      if (res.ok) {
   232→        toast.success('Requisition closed — inventory issued & stock updated')
   233→        setCompleteOpen(false)
   234→        onOpenChange(false)
   235→        onAction()
   236→      } else {
   237→        const d = await res.json()
   238→        toast.error(d.error || 'Failed to complete')
   239→      }
   240→    } catch { toast.error('Network error') } finally { setActionLoading(false) }
   241→  }
   242→
   243→  // ─── Save Signatures ────────────────────────────────────────────────────
   244→  async function handleSaveSignatures() {
   245→    if (!requisitionId) return
   246→    setSavingSignatures(true)
   247→    try {
   248→      const res = await fetch(`/api/requisitions/${requisitionId}`, {
   249→        method: 'PUT',
   250→        headers: { 'Content-Type': 'application/json' },
   251→        body: JSON.stringify({ issuedByName: issuedByName || null, receivedByName: receivedByName || null }),
   252→      })
   253→      if (res.ok) {
   254→        toast.success('Signatures updated')
   255→        setEditIssuedBy(false)
   256→        setEditReceivedBy(false)
   257→        fetchData()
   258→      } else {
   259→        const d = await res.json()
   260→        toast.error(d.error || 'Failed to update')
   261→      }
   262→    } catch { toast.error('Network error') } finally { setSavingSignatures(false) }
   263→  }
   264→
   265→  const openApproveDialog = () => {
   266→    if (!data) return
   267→    const init: Record<string, number> = {}
   268→    data.items.forEach((item) => { init[item.id] = item.requiredQty })
   269→    setApproveQtys(init)
   270→    setApproveOpen(true)
   271→  }
   272→
   273→  const isClosed = data?.status === 'CLOSED' || data?.status === 'COMPLETED'
   274→
   275→  return (
   276→    <>
   277→      <Dialog open={open} onOpenChange={onOpenChange}>
   278→        <DialogContent className="max-w-5xl max-h-[92vh] overflow-hidden flex flex-col">
   279→          <DialogHeader>
   280→            <DialogTitle className="text-center font-bold tracking-wider uppercase flex items-center justify-center gap-2">
   281→              <FileText className="size-5" />
   282→              Material Requisition
   283→            </DialogTitle>
   284→            <DialogDescription className="text-center">
   285→              {data?.requisitionNo && (
   286→                <span className="font-mono font-semibold text-foreground">{data.requisitionNo}</span>
   287→              )}
   288→            </DialogDescription>
   289→          </DialogHeader>
   290→
   291→          {loading ? (
   292→            <div className="flex-1 overflow-y-auto space-y-4 p-4">
   293→              <Skeleton className="h-20 w-full" />
   294→              <Skeleton className="h-60 w-full" />
   295→              <Skeleton className="h-24 w-full" />
   296→            </div>
   297→          ) : data ? (
   298→            <div className="flex-1 overflow-y-auto space-y-4">
   299→              {/* ─── Status Row ──────────────────────────────────────────── */}
   300→              <div className="flex items-center justify-between px-1">
   301→                <Badge className={`${statusColors[data.status] || statusColors.CANCELLED} text-xs font-semibold border-0 px-3 py-1`}>
   302→                  {data.status === 'CLOSED' ? 'Closed' : data.status.replace(/_/g, ' ')}
   303→                </Badge>
   304→                <span className="text-xs text-muted-foreground">
   305→                  {format(new Date(data.date), 'dd-MMM-yy')}
   306→                </span>
   307→              </div>
   308→
   309→              {/* ─── Header Info ─────────────────────────────────────────── */}
   310→              <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 p-4 border rounded-lg bg-muted/30 text-sm">
   311→                <div>
   312→                  <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-semibold">Department</p>
   313→                  <p className="font-medium mt-0.5">{data.department.name}</p>
   314→                  <p className="text-[10px] text-muted-foreground font-mono">{data.department.code}</p>
   315→                </div>
   316→                <div>
   317→                  <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-semibold">Project</p>
   318→                  <p className="font-medium mt-0.5">{data.project.name}</p>
   319→                  <p className="text-[10px] text-muted-foreground font-mono">{data.project.code}</p>
   320→                </div>
   321→                <div>
   322→                  <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-semibold">Employee</p>
   323→                  <p className="font-medium mt-0.5 flex items-center gap-1">
   324→                    <User className="size-3 text-muted-foreground" />
   325→                    {data.employeeName}
   326→                  </p>
   327→                </div>
   328→                <div>
   329→                  <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-semibold">Requested By</p>
   330→                  <p className="font-medium mt-0.5">{data.requestedByUser.name}</p>
   331→                  <p className="text-[10px] text-muted-foreground flex items-center gap-0.5">
   332→                    <Mail className="size-2.5" /> {data.requestedByUser.email}
   333→                  </p>
   334→                </div>
   335→              </div>
   336→
   337→              {/* ─── Items Table with Stock Info ──────────────────────────── */}
   338→              <div className="border rounded-lg overflow-hidden">
   339→                <div className="overflow-x-auto">
   340→                  <Table>
   341→                    <TableHeader>
   342→                      <TableRow className="bg-muted/50">
   343→                        <TableHead className="w-[50px] text-center font-semibold">S/No</TableHead>
   344→                        <TableHead className="font-semibold min-w-[160px]">Item Name</TableHead>
   345→                        <TableHead className="font-semibold">Item ID</TableHead>
   346→                        <TableHead className="font-semibold">Spec</TableHead>
   347→                        <TableHead className="w-[90px] text-center font-semibold">Req Qty</TableHead>
   348→                        <TableHead className="w-[90px] text-center font-semibold">Issued</TableHead>
   349→                        <TableHead className="w-[90px] text-center font-semibold">In Stock</TableHead>
   350→                        <TableHead className="w-[90px] text-center font-semibold">Ordered</TableHead>
   351→                        <TableHead className="w-[90px] text-center font-semibold">Left</TableHead>
   352→                        <TableHead className="font-semibold">Status</TableHead>
   353→                      </TableRow>
   354→                    </TableHeader>
   355→                    <TableBody>
   356→                      {data.items.map((item, idx) => {
   357→                        const stock = item.stockInfo
   358→                        const allIssued = item.issuedQty >= item.requiredQty
   359→                        const partialIssued = item.issuedQty > 0 && !allIssued
   360→
   361→                        return (
   362→                          <TableRow key={item.id}>
   363→                            <TableCell className="text-center font-mono text-sm text-muted-foreground">
   364→                              {idx + 1}
   365→                            </TableCell>
   366→                            <TableCell>
   367→                              <p className="font-medium text-sm">{item.product.name}</p>
   368→                              <p className="text-[10px] text-muted-foreground font-mono">{item.product.code}</p>
   369→                            </TableCell>
   370→                            <TableCell className="text-xs font-mono text-muted-foreground">
   371→                              {item.product.sku || item.product.code}
   372→                            </TableCell>
   373→                            <TableCell className="text-sm">
   374→                              {item.specDescription || '—'}
   375→                            </TableCell>
   376→                            <TableCell className="text-center font-semibold text-sm">
   377→                              {item.requiredQty} <span className="text-[10px] font-normal text-muted-foreground">{item.product.unit}</span>
   378→                            </TableCell>
   379→                            <TableCell className="text-center text-sm">
   380→                              {item.issuedQty > 0 ? (
   381→                                <span className="font-semibold text-green-600 dark:text-green-400">
   382→                                  {item.issuedQty}
   383→                                </span>
   384→                              ) : (
   385→                                <span className="text-muted-foreground">—</span>
   386→                              )}
   387→                            </TableCell>
   388→                            <TableCell className="text-center text-sm">
   389→                              <span className="flex items-center justify-center gap-1">
   390→                                <Warehouse className="size-3 text-muted-foreground" />
   391→                                <span className="font-medium">{stock?.totalInStock ?? '—'}</span>
   392→                              </span>
   393→                            </TableCell>
   394→                            <TableCell className="text-center text-sm">
   395→                              <span className="flex items-center justify-center gap-1">
   396→                                <Truck className="size-3 text-muted-foreground" />
   397→                                <span className="font-medium">{stock?.totalIssued ?? '—'}</span>
   398→                              </span>
   399→                            </TableCell>
   400→                            <TableCell className="text-center text-sm">
   401→                              <span className={`font-semibold ${
   402→                                (stock?.availableInStore ?? 0) > item.requiredQty ? 'text-green-600 dark:text-green-400' :
   403→                                (stock?.availableInStore ?? 0) === 0 ? 'text-red-600 dark:text-red-400' :
   404→                                'text-amber-600 dark:text-amber-400'
   405→                              }`}>
   406→                                {stock?.availableInStore ?? '—'}
   407→                              </span>
   408→                            </TableCell>
   409→                            <TableCell className="text-center">
   410→                              {isClosed ? (
   411→                                <Badge className="bg-gray-200 text-gray-700 dark:bg-gray-700 dark:text-gray-300 text-[10px] border-0 font-semibold px-2">
   412→                                  Closed
   413→                                </Badge>
   414→                              ) : allIssued ? (
   415→                                <Badge className="bg-green-100 text-green-700 dark:bg-green-900/30 dark:text-green-400 text-[10px] border-0 font-semibold px-2 flex items-center justify-center gap-0.5 w-fit mx-auto">
   416→                                  <PackageCheck className="size-3" />
   417→                                  Fulfilled
   418→                                </Badge>
   419→                              ) : partialIssued ? (
   420→                                <Badge className="bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400 text-[10px] border-0 font-semibold px-2">
   421→                                  Partial
   422→                                </Badge>
   423→                              ) : (
   424→                                <Badge className="bg-blue-100 text-blue-700 dark:bg-blue-900/30 dark:text-blue-400 text-[10px] border-0 font-semibold px-2">
   425→                                  Pending
   426→                                </Badge>
   427→                              )}
   428→                            </TableCell>
   429→                          </TableRow>
   430→                        )
   431→                      })}
   432→                    </TableBody>
   433→                  </Table>
   434→                </div>
   435→              </div>
   436→
   437→              {/* ─── Rejection Remarks ────────────────────────────────────── */}
   438→              {data.status === 'REJECTED' && data.remarks && (
   439→                <div className="rounded-lg border border-red-200 bg-red-50 dark:bg-red-900/10 p-3">
   440→                  <p className="text-xs font-semibold text-red-600 flex items-center gap-1 mb-1">
   441→                    <XCircle className="size-3" /> Rejection Reason
   442→                  </p>
   443→                  <p className="text-sm text-red-800 dark:text-red-300">{data.remarks}</p>
   444→                </div>
   445→              )}
   446→
   447→              {/* ─── Closed Banner ─────────────────────────────────────────── */}
   448→              {isClosed && (
   449→                <div className="rounded-lg border border-emerald-200 bg-emerald-50 dark:bg-emerald-900/10 p-3 flex items-center gap-2">
   450→                  <CheckCheck className="size-5 text-emerald-600" />
   451→                  <div>
   452→                    <p className="text-sm font-semibold text-emerald-700 dark:text-emerald-400">Requisition Closed</p>
   453→                    <p className="text-xs text-emerald-600 dark:text-emerald-500">
   454→                      All items have been issued. Stock has been automatically deducted.
   455→                      {data.completedAt && ` Closed: ${format(new Date(data.completedAt), "MMM dd, yyyy 'at' HH:mm")}`}
   456→                    </p>
   457→                  </div>
   458→                </div>
   459→              )}
   460→
   461→              {/* ─── Issued By / Received By (Signature Sections) ─────────── */}
   462→              <div className="grid grid-cols-2 gap-4">
   463→                <div className="border-2 border-dashed rounded-lg p-4 min-h-[100px]">
   464→                  <div className="flex items-center justify-between mb-2">
   465→                    <Label className="text-xs uppercase tracking-wider font-bold text-muted-foreground">
   466→                      Issued By
   467→                    </Label>
   468→                    {(data.status === 'APPROVED' || data.status === 'PARTIAL_APPROVED' || isClosed) && canEdit && !editIssuedBy && (
   469→                      <Button variant="ghost" size="icon" className="size-6" onClick={() => setEditIssuedBy(true)}>
   470→                        <PenLine className="size-3" />
   471→                      </Button>
   472→                    )}
   473→                  </div>
   474→                  {editIssuedBy ? (
   475→                    <div className="flex gap-2">
   476→                      <Input
   477→                        value={issuedByName}
   478→                        onChange={(e) => setIssuedByName(e.target.value)}
   479→                        placeholder="Enter issuer name"
   480→                        className="h-8"
   481→                        autoFocus
   482→                      />
   483→                      <Button size="sm" className="h-8" onClick={handleSaveSignatures} disabled={savingSignatures}>
   484→                        {savingSignatures ? <Loader2 className="size-3 animate-spin" /> : 'Save'}
   485→                      </Button>
   486→                    </div>
   487→                  ) : (
   488→                    <p className="text-lg font-semibold mt-4">
   489→                      {data.issuedByName || (
   490→                        <span className="text-muted-foreground font-normal text-base italic">Signature / Name</span>
   491→                      )}
   492→                    </p>
   493→                  )}
   494→                </div>
   495→                <div className="border-2 border-dashed rounded-lg p-4 min-h-[100px]">
   496→                  <div className="flex items-center justify-between mb-2">
   497→                    <Label className="text-xs uppercase tracking-wider font-bold text-muted-foreground">
   498→                      Received By
   499→                    </Label>
   500→                    {(data.status === 'APPROVED' || data.status === 'PARTIAL_APPROVED' || isClosed) && canEdit && !editReceivedBy && (
   501→                      <Button variant="ghost" size="icon" className="size-6" onClick={() => setEditReceivedBy(true)}>
   502→                        <PenLine className="size-3" />
   503→                      </Button>
   504→                    )}
   505→                  </div>
   506→                  {editReceivedBy ? (
   507→                    <div className="flex gap-2">
   508→                      <Input
   509→                        value={receivedByName}
   510→                        onChange={(e) => setReceivedByName(e.target.value)}
   511→                        placeholder="Enter receiver name"
   512→                        className="h-8"
   513→                        autoFocus
   514→                      />
   515→                      <Button size="sm" className="h-8" onClick={handleSaveSignatures} disabled={savingSignatures}>
   516→                        {savingSignatures ? <Loader2 className="size-3 animate-spin" /> : 'Save'}
   517→                      </Button>
   518→                    </div>
   519→                  ) : (
   520→                    <p className="text-lg font-semibold mt-4">
   521→                      {data.receivedByName || (
   522→                        <span className="text-muted-foreground font-normal text-base italic">Signature / Name</span>
   523→                      )}
   524→                    </p>
   525→                  )}
   526→                </div>
   527→              </div>
   528→
   529→              {/* ─── Action Buttons ────────────────────────────────────────── */}
   530→              {data.status === 'PENDING' && (canApprove || canReject) && (
   531→                <div className="flex items-center gap-2 pt-2">
   532→                  {canApprove && (
   533→                    <Button size="sm" className="bg-green-600 hover:bg-green-700 flex-1" onClick={openApproveDialog}>
   534→                      <CheckCircle className="mr-2 size-4" /> Approve
   535→                    </Button>
   536→                  )}
   537→                  {canReject && (
   538→                    <Button size="sm" variant="destructive" className="flex-1" onClick={() => { setRejectRemarks(''); setRejectOpen(true) }}>
   539→                      <XCircle className="mr-2 size-4" /> Reject
   540→                    </Button>
   541→                  )}
   542→                </div>
   543→              )}
   544→
   545→              {(data.status === 'APPROVED' || data.status === 'PARTIAL_APPROVED') && canEdit && (
   546→                <div className="pt-2">
   547→                  <Button size="sm" className="w-full bg-emerald-600 hover:bg-emerald-700" onClick={() => setCompleteOpen(true)}>
   548→                    <CheckCheck className="mr-2 size-4" /> Complete & Issue Inventory
   549→                  </Button>
   550→                  <p className="text-xs text-muted-foreground mt-1.5">
   551→                    This will issue all approved items to {data.department.name} — {data.project.name} and automatically deduct stock.
   552→                  </p>
   553→                </div>
   554→              )}
   555→
   556→              {/* ─── Timestamp ─────────────────────────────────────────────── */}
   557→              <p className="text-[10px] text-muted-foreground text-center pt-1">
   558→                Created: {format(new Date(data.createdAt), "MMM dd, yyyy 'at' HH:mm")}
   559→                {data.approvedAt && ` · Approved: ${format(new Date(data.approvedAt), "MMM dd, yyyy 'at' HH:mm")}`}
   560→                {data.completedAt && ` · Closed: ${format(new Date(data.completedAt), "MMM dd, yyyy 'at' HH:mm")}`}
   561→              </p>
   562→            </div>
   563→          ) : (
   564→            <div className="flex-1 flex flex-col items-center justify-center py-12">
   565→              <FileText className="size-12 text-muted-foreground/50 mb-4" />
   566→              <p className="text-sm text-muted-foreground">Requisition not found</p>
   567→            </div>
   568→          )}
   569→        </DialogContent>
   570→      </Dialog>
   571→
   572→      {/* ─── Approve Dialog ────────────────────────────────────────────────── */}
   573→      <Dialog open={approveOpen} onOpenChange={setApproveOpen}>
   574→        <DialogContent className="max-w-2xl max-h-[80vh] overflow-y-auto">
   575→          <DialogHeader>
   576→            <DialogTitle className="flex items-center gap-2 text-green-600">
   577→              <CheckCircle className="size-5" /> Approve Requisition
   578→            </DialogTitle>
   579→            <DialogDescription>
   580→              Review and adjust quantities for each item before approving.
   581→            </DialogDescription>
   582→          </DialogHeader>
   583→          {data && (
   584→            <div className="border rounded-lg overflow-hidden">
   585→              <Table>
   586→                <TableHeader>
   587→                  <TableRow className="bg-muted/50">
   588→                    <TableHead>Item</TableHead>
   589→                    <TableHead className="text-center">Requested</TableHead>
   590→                    <TableHead className="text-center">Approve Qty</TableHead>
   591→                    <TableHead className="text-center">Available</TableHead>
   592→                  </TableRow>
   593→                </TableHeader>
   594→                <TableBody>
   595→                  {data.items.map((item) => (
   596→                    <TableRow key={item.id}>
   597→                      <TableCell className="text-sm">{item.product.name}</TableCell>
   598→                      <TableCell className="text-center text-sm font-medium">{item.requiredQty} {item.product.unit}</TableCell>
   599→                      <TableCell className="text-center">
   600→                        <Input
   601→                          type="number"
   602→                          className="w-24 h-8 text-center mx-auto"
   603→                          min={0}
   604→                          max={item.requiredQty}
   605→                          value={approveQtys[item.id] ?? item.requiredQty}
   606→                          onChange={(e) =>
   607→                            setApproveQtys((prev) => ({
   608→                              ...prev,
   609→                              [item.id]: Math.min(Number(e.target.value) || 0, item.requiredQty),
   610→                            }))
   611→                          }
   612→                        />
   613→                      </TableCell>
   614→                      <TableCell className="text-center text-sm font-medium">
   615→                        <span className={
   616→                          (item.stockInfo?.availableInStore ?? 0) >= item.requiredQty
   617→                            ? 'text-green-600 dark:text-green-400'
   618→                            : 'text-red-600 dark:text-red-400'
   619→                        }>
   620→                          {item.stockInfo?.availableInStore ?? '—'}
   621→                        </span>
   622→                      </TableCell>
   623→                    </TableRow>
   624→                  ))}
   625→                </TableBody>
   626→              </Table>
   627→            </div>
   628→          )}
   629→          <DialogFooter>
   630→            <Button variant="outline" onClick={() => setApproveOpen(false)}>Cancel</Button>
   631→            <Button className="bg-green-600 hover:bg-green-700" disabled={actionLoading} onClick={handleApprove}>
   632→              {actionLoading && <Loader2 className="mr-2 size-4 animate-spin" />}
   633→              Approve
   634→            </Button>
   635→          </DialogFooter>
   636→        </DialogContent>
   637→      </Dialog>
   638→
   639→      {/* ─── Reject Dialog ─────────────────────────────────────────────────── */}
   640→      <Dialog open={rejectOpen} onOpenChange={setRejectOpen}>
   641→        <DialogContent className="max-w-md">
   642→          <DialogHeader>
   643→            <DialogTitle className="flex items-center gap-2 text-red-600">
   644→              <XCircle className="size-5" /> Reject Requisition
   645→            </DialogTitle>
   646→            <DialogDescription>Provide a reason for rejecting this requisition.</DialogDescription>
   647→          </DialogHeader>
   648→          <Textarea
   649→            placeholder="Enter rejection reason..."
   650→            value={rejectRemarks}
   651→            onChange={(e) => setRejectRemarks(e.target.value)}
   652→            rows={3}
   653→            autoFocus
   654→          />
   655→          <DialogFooter>
   656→            <Button variant="outline" onClick={() => setRejectOpen(false)}>Cancel</Button>
   657→            <Button variant="destructive" disabled={actionLoading} onClick={handleReject}>
   658→              {actionLoading && <Loader2 className="mr-2 size-4 animate-spin" />}
   659→              Reject
   660→            </Button>
   661→          </DialogFooter>
   662→        </DialogContent>
   663→      </Dialog>
   664→
   665→      {/* ─── Complete Confirmation ──────────────────────────────────────────── */}
   666→      <AlertDialog open={completeOpen} onOpenChange={setCompleteOpen}>
   667→        <AlertDialogContent>
   668→          <AlertDialogHeader>
   669→            <AlertDialogTitle className="flex items-center gap-2">
   670→              <CheckCheck className="size-5 text-emerald-600" /> Complete Requisition
   671→            </AlertDialogTitle>
   672→            <AlertDialogDescription>
   673→              This will issue all approved items to <strong>{data?.department.name}</strong> — <strong>{data?.project.name}</strong>.<br />
   674→              Inventory will be automatically deducted from Stock.
   675→              The requisition will be marked as <strong>Closed</strong>.
   676→            </AlertDialogDescription>
   677→          </AlertDialogHeader>
   678→          <AlertDialogFooter>
   679→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   680→            <AlertDialogAction onClick={handleComplete} disabled={actionLoading} className="bg-emerald-600 hover:bg-emerald-700">
   681→              {actionLoading && <Loader2 className="mr-2 size-4 animate-spin" />}
   682→              Complete & Close
   683→            </AlertDialogAction>
   684→          </AlertDialogFooter>
   685→        </AlertDialogContent>
   686→      </AlertDialog>
   687→    </>
   688→  )
   689→}
