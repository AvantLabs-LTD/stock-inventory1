   213→  // Export CSV
   214→  async function handleExport() {
   215→    setExporting(true)
   216→    try {
   217→      const params = new URLSearchParams({ reportType })
   218→      if (dateFrom) params.set('dateFrom', dateFrom)
   219→      if (dateTo) params.set('dateTo', dateTo)
   220→      if (categoryId) params.set('categoryId', categoryId)
   221→      if (departmentId) params.set('departmentId', departmentId)
   222→      if (projectId) params.set('projectId', projectId)
   223→      if (status) params.set('status', status)
   224→      if (productId) params.set('productId', productId)
   225→      if (supplierId) params.set('supplierId', supplierId)
   226→
   227→      const res = await fetch(`/api/reports/export/excel?${params}`)
   228→      if (!res.ok) {
   229→        toast.error('Export failed')
   230→        return
   231→      }
   232→      const blob = await res.blob()
   233→      const url = URL.createObjectURL(blob)
   234→      const a = document.createElement('a')
   235→      a.href = url
   236→      const disposition = res.headers.get('Content-Disposition')
   237→      const filename = disposition
   238→        ? disposition.split('filename=')[1]?.replace(/"/g, '')
   239→        : `${reportType}.csv`
   240→      a.download = filename
   241→      document.body.appendChild(a)
   242→      a.click()
   243→      document.body.removeChild(a)
   244→      URL.revokeObjectURL(url)
   245→      toast.success('Report exported successfully')
   246→    } catch {
   247→      toast.error('Export failed')
   248→    } finally {
   249→      setExporting(false)
   250→    }
   251→  }
