   380→  function validateFile(file: File): string | null {
   381→    const validExts = ['.xlsx', '.xls', '.csv']
   382→    const validMimes = [
   383→      'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
   384→      'application/vnd.ms-excel',
   385→      'text/csv',
   386→      'application/csv',
   387→      'application/octet-stream',
   388→    ]
   389→    const ext = file.name.substring(file.name.lastIndexOf('.')).toLowerCase()
   390→    if (!validExts.includes(ext) && !validMimes.includes(file.type)) {
   391→      return 'Only .xlsx, .xls, or .csv files are supported'
   392→    }
   393→    if (file.size > 50 * 1024 * 1024) {
   394→      return 'File is too large. Maximum size is 50 MB.'
   395→    }
   396→    if (file.size === 0) {
   397→      return 'File is empty. Please select a valid file.'
   398→    }
   399→    return null
   400→  }
