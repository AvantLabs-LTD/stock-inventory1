   158→  const [supplierId, setSupplierId] = useState('')
   159→  const [purchaseRef, setPurchaseRef] = useState('')

   247→              parentProductId: data.parentProductId,
   248→              variantName: data.variantName,
   249→              category: data.category,
   250→              supplier: data.supplier,

   254→        })

   256→    }

   258→
   259→  // ─── Reset on open for create mode ────────────────────────────────────
   260→  useEffect(() => {
   261→    if (open && !editEntry) {
   262→      setProductSearch('')
   263→      setProductResults([])
   264→      setSelectedProduct(null)
   265→      setShowProductCreate(false)
   266→      setWarehouse('')
   267→      setStorageLocation('')
   268→      setOpeningQty(0)
   269→      setUnitCost(0)
   270→      setBatchNumber('')
   271→      setSerialNumber('')
   272→      setExpiryDate('')
   273→      setSupplierId('')
   274→      setPurchaseRef('')
   275→      setInvoiceNumber('')
   276→      setReceivedDate(format(new Date(), 'yyyy-MM-dd'))
   277→      setOpeningDate(format(new Date(), 'yyyy-MM-dd'))
   278→      setRemarks('')
   279→      setInternalNotes('')
   280→      setNewProductName('')
   281→      setNewProductCategory('')
   282→      setNewProductUnit('pcs')
   283→      setNewProductVariantName('')
   284→      setNewProductParentId('')
   285→      setProjectQtys(
   286→        projectFields.map((pf) => ({
   287→          projectFieldId: pf.id,
   288→          requiredQty: 0,
   289→          batchQty: 0,

   299→  }, [open, editEntry, projectFields])
   300→

   302→  useEffect(() => {
   303→    if (!open) return
   304→    const fetchRefs = async () => {
   305→      try {
   306→        const [sRes, cRes] = await Promise.all([
   307→          fetch('/api/suppliers?limit=500'),
   308→          fetch('/api/categories?limit=500'),
   309→        ])
   310→        if (sRes.ok) {
   311→          const sJson = await sRes.json()
   312→          setSuppliers(sJson.data || [])
   313→        }
   314→        if (cRes.ok) {
   315→          const cJson = await cRes.json()
   316→          setCategories(cJson.data || [])
   317→        }
   318→      } catch {
   319→        // ignore
   320→      }
   321→    }
   322→    fetchRefs()
   323→  }, [open])
   324→
   325→  // ─── Product Search ────────────────────────────────────────────────────
   326→  const handleProductSearch = useCallback(async (query: string) => {
   327→    setProductSearch(query)
   328→    setSelectedProduct(null)
   329→    if (!query || query.length < 2) {
   330→      setProductResults([])

   335→      const res = await fetch(`/api/products?search=${encodeURIComponent(query)}&limit=20&status=ACTIVE`)

   337→      const json = await res.json()
   338→      setProductResults(json.data || [])
   339→    } catch {
   340→      setProductResults([])
   341→    } finally {
   342→      setProductSearchLoading(false)
   343→    }
   344→  }, [])
   345→
   346→  const handleSelectProduct = useCallback((product: ProductItem) => {
   347→    setSelectedProduct(product)
   348→    setProductSearch(product.name)
   349→    setProductResults([])
   350→    setShowProductCreate(false)
   351→    // Auto-set supplier if product has one
   352→    if (product.supplierId) {
   353→      setSupplierId(product.supplierId)
   354→    }
   355→    // Auto-set unit
   356→    if (!editEntry) {
   357→      setUnitCost(product.supplierId ? 0 : 0) // keep existing cost
   358→    }
   359→  }, [editEntry])
   360→

   363→    if (!newProductName.trim()) return
   364→    setCreatingProduct(true)
   365→    try {
   366→      const body: Record<string, unknown> = {
   367→        name: newProductName.trim(),
   368→        unit: newProductUnit,
   369→        categoryId: newProductCategory || undefined,
   370→        variantName: newProductVariantName || undefined,

   374→        method: 'POST',
   375→        headers: { 'Content-Type': 'application/json' },
   376→        body: JSON.stringify(body),
   377→      })
   378→      if (!res.ok) {
   379→        const err = await res.json()
   380→        throw new Error(err.error || 'Failed to create product')
   381→      }
   382→      const product = await res.json()
   383→      toast.success(`Product "${product.name}" created`)
   384→      handleSelectProduct({
   385→        id: product.id,
   386→        name: product.name,
   387→        code: product.code,
   388→        sku: product.sku,
   389→        unit: product.unit,
   390→        status: 'ACTIVE',
   391→        categoryId: product.categoryId,
   392→        brand: null,
   393→        supplierId: null,
   394→        parentProductId: product.parentProductId,
   395→        variantName: product.variantName,
   396→        category: null,
   397→        supplier: null,
   398→        parent: null,
   399→      })
   400→      setShowProductCreate(false)
   401→      setNewProductName('')
   402→    } catch (err) {
   403→      toast.error(err instanceof Error ? err.message : 'Failed to create product')
   404→    } finally {
   405→      setCreatingProduct(false)
   406→    }
   407→  }, [newProductName, newProductUnit, newProductCategory, newProductVariantName, newProductParentId, handleSelectProduct])
   408→

   410→  const updateProjectQty = useCallback((idx: number, field: keyof ProjectQtyRow, value: number | string) => {

   412→      const next = [...prev]
   413→      next[idx] = { ...next[idx], [field]: value }

   422→

   424→  const handleSubmit = useCallback(async () => {
   425→    if (!selectedProduct && !editEntry) {
   426→      toast.error('Please select a product')
   427→      return
   428→    }
   429→    if (!isEdit && openingQty <= 0) {
   430→      toast.error('Opening quantity must be greater than 0')

   433→
   434→    setSubmitting(true)

   436→      if (isEdit && editEntry) {
   437→        // Edit mode - only limited fields

   439→          remarks,
   440→          internalNotes,
   441→          batchNumber: batchNumber || null,
   442→          serialNumber: serialNumber || null,
   443→          expiryDate: expiryDate || null,
   444→          storageLocation: storageLocation || null,

   464→          storageLocation: storageLocation || null,
   465→          batchNumber: batchNumber || null,
   466→          serialNumber: serialNumber || null,
   467→          expiryDate: expiryDate || null,
   468→          supplierId: supplierId || null,
   469→          purchaseReference: purchaseRef || null,
   470→          invoiceNumber: invoiceNumber || null,
   471→          receivedDate: receivedDate || null,

   474→          internalNotes: internalNotes || null,
   475→          projectQtys: projectQtys

   477→            .map((pq) => ({
   478→              projectFieldId: pq.projectFieldId,
   479→              requiredQty: pq.requiredQty,
   480→              batchQty: pq.batchQty,
   481→              reservedQty: pq.reservedQty,
   482→              consumedQty: pq.consumedQty,
   483→              orderedQty: pq.orderedQty,
   484→              toBeUsed: pq.toBeUsed,
   485→              remarks: pq.remarks || null,
   486→            })),
   487→        }
   488→
   489→        const res = await fetch('/api/opening-stock', {
   490→          method: 'POST',
   491→          headers: { 'Content-Type': 'application/json' },
   492→          body: JSON.stringify(body),
   493→        })
   494→        if (!res.ok) {
   495→          const err = await res.json()
   496→          throw new Error(err.error || 'Failed to create')
   497→        }
   498→        toast.success('Opening stock created successfully')

   500→
   501→      onOpenChange(false)
   502→      onSaved()
   503→    } catch (err) {
   504→      toast.error(err instanceof Error ? err.message : 'Operation failed')
   505→    } finally {
   506→      setSubmitting(false)
   507→    }
   508→  }, [
   509→    selectedProduct,
   510→    editEntry,
   511→    isEdit,
   512→    openingQty,
   513→    unitCost,
   514→    warehouse,
   515→    storageLocation,
   516→    batchNumber,
   517→    serialNumber,
   518→    expiryDate,
   519→    supplierId,
   520→    purchaseRef,
   521→    invoiceNumber,
   522→    receivedDate,
   523→    openingDate,
   524→    remarks,
   525→    internalNotes,
   526→    projectQtys,
   527→    onOpenChange,
   528→    onSaved,
   529→  ])
   530→
   531→  // ─── Section Component ──────────────────────────────────────────────────
   532→  const SectionHeader = ({
   533→    sectionKey,
   534→    title,
   535→    subtitle,
   536→    badge,

   546→        {subtitle && <span className="text-xs text-muted-foreground">{subtitle}</span>}
   547→        {badge && (
   548→          <Badge variant="secondary" className="text-xs">
   549→            {badge}
   550→          </Badge>
   551→        )}
   552→      </div>
   553→      {sectionsOpen[sectionKey] ? (
   554→        <ChevronUp className="h-4 w-4 text-muted-foreground" />
   555→      ) : (
   556→        <ChevronDown className="h-4 w-4 text-muted-foreground" />
   557→      )}
   558→    </CollapsibleTrigger>
   559→  )
   560→
   561→  return (
   562→    <Dialog open={open} onOpenChange={onOpenChange}>
   563→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-hidden flex flex-col">
   564→        <DialogHeader>
   565→          <DialogTitle>{isEdit ? 'Edit Opening Stock' : 'Add Opening Stock'}</DialogTitle>
   566→          <DialogDescription>
   567→            {isEdit ? 'Update inventory metadata and project quantities.' : 'Create a new opening stock entry for inventory.'}
   568→          </DialogDescription>
   569→        </DialogHeader>
   570→
   571→        <div className="flex-1 overflow-y-auto space-y-3 pr-1">

   576→              title="Product Information"
   577→              subtitle="Search and select the product"

   580→            <CollapsibleContent className="space-y-4 pt-3 px-1">

   589→                        if (isEdit) return // no change in edit mode
   590→                        handleProductSearch(e.target.value)
   591→                      }}
   592→                      className="pl-9"
   593→                      disabled={isEdit}
   594→                    />
   595→                    {productSearchLoading && (
   596→                      <Loader2 className="absolute right-3 top-1/2 h-4 w-4 -translate-y-1/2 animate-spin text-muted-foreground" />
   597→                    )}
   598→                  </div>
   599→
   600→                  {/* Search Results Dropdown */}
   601→                  {productResults.length > 0 && !selectedProduct && (
   602→                    <div className="rounded-md border bg-background shadow-lg max-h-48 overflow-y-auto">
   603→                      {productResults.map((p) => (
   604→                        <button
   605→                          key={p.id}
   606→                          className="flex w-full items-center justify-between px-3 py-2 text-sm hover:bg-muted/50 transition-colors"
   607→                          onClick={() => handleSelectProduct(p)}

   609→                          <div>
   610→                            <p className="font-medium">{p.name}</p>
   611→                            <p className="text-xs text-muted-foreground">
   612→                              {p.code} | {p.sku} | {p.category?.name || 'No Category'}
   613→                              {p.parentProductId && ` (Variant: ${p.variantName || 'N/A'})`}
   614→                            </p>
   615→                          </div>
   616→                          <Badge variant="outline" className="text-xs">
   617→                            {p.unit}
   618→                          </Badge>
   619→                        </button>
   620→                      ))}
   621→                    </div>
   622→                  )}
   623→
   624→                  {/* Product not found + create prompt */}
   625→                  {productSearch.length >= 2 && !productSearchLoading && productResults.length === 0 && !selectedProduct && (
   626→                    <div className="flex items-center gap-2 rounded-md border border-dashed p-3">
   627→                      <AlertCircle className="h-4 w-4 text-amber-500" />
   628→                      <span className="text-sm text-muted-foreground">
   629→                        Product not found.{' '}
   630→                        <button
   631→                          className="text-primary font-medium hover:underline"
   632→                          onClick={() => setShowProductCreate(true)}
   633→                        >
   634→                          Create Product?
   635→                        </button>
   636→                      </span>
   637→                    </div>
   638→                  )}
   639→
   640→                  {/* Selected Product Info */}
   641→                  {selectedProduct && (
   642→                    <div className="rounded-md border bg-muted/30 p-3 space-y-2">
   643→                      <div className="flex items-center gap-2">
   644→                        <div className="flex h-8 w-8 items-center justify-center rounded-md bg-primary/10 text-xs font-semibold text-primary">
   645→                          {selectedProduct.name.charAt(0).toUpperCase()}

   647→                        <div>
   648→                          <p className="font-medium text-sm">{selectedProduct.name}</p>
   649→                          <p className="text-xs text-muted-foreground">{selectedProduct.code} | {selectedProduct.sku}</p>
   650→                        </div>
   651→                        {!isEdit && (
   652→                          <Button variant="ghost" size="icon" className="ml-auto h-7 w-7" onClick={() => { setSelectedProduct(null); setProductSearch('') }}>
   653→                            <X className="h-3.5 w-3.5" />
   654→                          </Button>
   655→                        )}
   656→                      </div>
   657→                      <div className="grid grid-cols-3 gap-2 text-xs">
   658→                        <div>
   659→                          <span className="text-muted-foreground">Category: </span>
   660→                          <span className="font-medium">{selectedProduct.category?.name || '—'}</span>
   661→                        </div>
   662→                        <div>
   663→                          <span className="text-muted-foreground">Brand: </span>
   664→                          <span className="font-medium">{selectedProduct.brand || '—'}</span>
   665→                        </div>
   666→                        <div>
   667→                          <span className="text-muted-foreground">Unit: </span>
   668→                          <span className="font-medium">{selectedProduct.unit}</span>
   669→                        </div>
   670→                      </div>
   671→                      {selectedProduct.parentProductId && (
   672→                        <p className="text-xs text-muted-foreground">
   673→                          Variant of: <span className="font-medium">{selectedProduct.parent?.name || 'Unknown'}</span>
   674→                          {selectedProduct.variantName && ` (${selectedProduct.variantName})`}
   675→                        </p>
   676→                      )}
   677→                    </div>
   678→                  )}
   679→                </>
   680→              ) : (
   681→                /* Quick Create Product */
   682→                <div className="space-y-3 rounded-md border p-4">
   683→                  <div className="flex items-center justify-between">
   684→                    <h4 className="text-sm font-semibold">Create New Product</h4>
   685→                    <Button variant="ghost" size="sm" onClick={() => setShowProductCreate(false)}>
   686→                      <X className="mr-1 h-3.5 w-3.5" />
   687→                      Cancel
   688→                    </Button>
   689→                  </div>
   690→                  <div className="grid grid-cols-2 gap-3">
   691→                    <div className="col-span-2">
   692→                      <Label className="text-xs">Product Name *</Label>
   693→                      <Input
   694→                        value={newProductName}
   695→                        onChange={(e) => setNewProductName(e.target.value)}
   696→                        placeholder="Enter product name"
   697→                      />
   698→                    </div>
   699→                    <div>
   700→                      <Label className="text-xs">Category</Label>
   701→                      <Select value={newProductCategory} onValueChange={setNewProductCategory}>
   702→                        <SelectTrigger>
   703→                          <SelectValue placeholder="Select category" />
   704→                        </SelectTrigger>
   705→                        <SelectContent>
   706→                          <SelectItem value="none">No Category</SelectItem>
   707→                          {categories.map((c) => (
   708→                            <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   709→                          ))}
   710→                        </SelectContent>
   711→                      </Select>
   712→                    </div>
   713→                    <div>
   714→                      <Label className="text-xs">Unit</Label>
   715→                      <Select value={newProductUnit} onValueChange={setNewProductUnit}>
   716→                        <SelectTrigger>
   717→                          <SelectValue />
   718→                        </SelectTrigger>
   719→                        <SelectContent>
   720→                          {['pcs', 'kg', 'm', 'mtr', 'pcs', 'set', 'lot', 'box', 'roll', 'pack'].map((u) => (
   721→                            <SelectItem key={u} value={u}>{u}</SelectItem>
   722→                          ))}
   723→                        </SelectContent>
   724→                      </Select>
   725→                    </div>
   726→                    <div>
   727→                      <Label className="text-xs">Variant Name (optional)</Label>
   728→                      <Input
   729→                        value={newProductVariantName}
   730→                        onChange={(e) => setNewProductVariantName(e.target.value)}
   731→                        placeholder="e.g., 10K, 100nF"
   732→                      />
   733→                    </div>
   734→                  </div>
   735→                  <Button onClick={handleQuickCreate} disabled={!newProductName.trim() || creatingProduct}>
   736→                    {creatingProduct && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   737→                    <Plus className="mr-2 h-4 w-4" />
   738→                    Create Product
   739→                  </Button>
   740→                </div>
   741→              )}
   742→            </CollapsibleContent>
   743→          </Collapsible>
   744→
   745→          {/* ─── 2. Inventory Information ───────────────────────────────── */}
   746→          <Collapsible open={sectionsOpen.inventory} onOpenChange={() => toggleSection('inventory')}>
   747→            <SectionHeader sectionKey="inventory" title="Inventory Information" subtitle="Warehouse, quantities, and costs" />
   748→            <CollapsibleContent className="space-y-4 pt-3 px-1">
   749→              <div className="grid grid-cols-2 gap-4">
   750→                <div>
   751→                  <Label className="text-xs">Warehouse</Label>
   752→                  <Select value={warehouse} onValueChange={setWarehouse}>
   753→                    <SelectTrigger>
   754→                      <SelectValue placeholder="Select warehouse" />
   755→                    </SelectTrigger>
   756→                    <SelectContent>
   757→                      {WAREHOUSES.map((w) => (
   758→                        <SelectItem key={w.value} value={w.value}>{w.label}</SelectItem>
   759→                      ))}
   760→                    </SelectContent>
   761→                  </Select>
   762→                </div>
   763→                <div>
   764→                  <Label className="text-xs">Storage Location</Label>
   765→                  <Input value={storageLocation} onChange={(e) => setStorageLocation(e.target.value)} placeholder="e.g., Shelf A-1, Bin 12" />
   766→                </div>
   767→                <div>
   768→                  <Label className="text-xs">
   769→                    Opening Quantity {isEdit ? '' : '*'}
   770→                  </Label>
   771→                  <Input
   772→                    type="number"
   773→                    value={openingQty || ''}
   774→                    onChange={(e) => setOpeningQty(Number(e.target.value) || 0)}
   775→                    placeholder="0"
   776→                    disabled={isEdit}
   777→                    min={0}
   778→                  />
   779→                  {isEdit && (
   780→                    <p className="text-xs text-muted-foreground mt-1">Cannot be edited. Use stock adjustments.</p>
   781→                  )}
   782→                </div>
   783→                <div>
   784→                  <Label className="text-xs">Unit Cost</Label>
   785→                  <Input
   786→                    type="number"
   787→                    value={unitCost || ''}
   788→                    onChange={(e) => setUnitCost(Number(e.target.value) || 0)}
   789→                    placeholder="0.00"
   790→                    disabled={isEdit}
   791→                    min={0}
   792→                    step={0.01}
   793→                  />
   794→                  {isEdit && (
   795→                    <p className="text-xs text-muted-foreground mt-1">Cannot be edited after creation.</p>
   796→                  )}
   797→                </div>
   798→              </div>
   799→
   800→              {/* Inventory Value */}
   801→              <div className="rounded-md bg-muted/50 border p-3">
   802→                <div className="flex items-center justify-between">
   803→                  <span className="text-sm text-muted-foreground">Calculated Inventory Value</span>
   804→                  <span className="text-lg font-bold text-primary">
   805→                    ${inventoryValue.toFixed(2)}
   806→                  </span>
   807→                </div>
   808→              </div>
   809→
   810→              <div className="grid grid-cols-2 gap-4">
   811→                <div>
   812→                  <Label className="text-xs">Batch Number</Label>
   813→                  <Input value={batchNumber} onChange={(e) => setBatchNumber(e.target.value)} placeholder="e.g., BATCH-001" />
   814→                </div>
   815→                <div>
   816→                  <Label className="text-xs">Serial Number</Label>
   817→                  <Input value={serialNumber} onChange={(e) => setSerialNumber(e.target.value)} placeholder="e.g., SN-001" />
   818→                </div>
   819→                <div>
   820→                  <Label className="text-xs">Expiry Date</Label>
   821→                  <Input type="date" value={expiryDate} onChange={(e) => setExpiryDate(e.target.value)} />
   822→                </div>
   823→              </div>
   824→            </CollapsibleContent>
   825→          </Collapsible>
   826→
   827→          {/* ─── 3. Supplier Information ────────────────────────────────── */}
   828→          <Collapsible open={sectionsOpen.supplier} onOpenChange={() => toggleSection('supplier')}>
   829→            <SectionHeader sectionKey="supplier" title="Supplier Information" subtitle="Supplier, purchase reference, invoice" />
   830→            <CollapsibleContent className="space-y-4 pt-3 px-1">
   831→              <div className="grid grid-cols-2 gap-4">
   832→                <div>
   833→                  <Label className="text-xs">Supplier</Label>
   834→                  <Select value={supplierId} onValueChange={(v) => setSupplierId(v === 'none' ? '' : v)}>
   835→                    <SelectTrigger>
   836→                      <SelectValue placeholder="Select supplier" />
   837→                    </SelectTrigger>
   838→                    <SelectContent>
   839→                      <SelectItem value="none">No Supplier</SelectItem>
   840→                      {suppliers.map((s) => (
   841→                        <SelectItem key={s.id} value={s.id}>{s.name}</SelectItem>
   842→                      ))}
   843→                    </SelectContent>
   844→                  </Select>
   845→                </div>
   846→                <div>
   847→                  <Label className="text-xs">Purchase Reference</Label>

   850→                <div>
   851→                  <Label className="text-xs">Invoice Number</Label>

   855→                  <Label className="text-xs">Received Date</Label>

   860→          </Collapsible>
   861→
   862→          {/* ─── 4. Project Quantities ─────────────────────────────────── */}
   863→          {projectFields.length > 0 && (
   864→            <Collapsible open={sectionsOpen.projects} onOpenChange={() => toggleSection('projects')}>

   867→                title="Project Quantities"
   868→                subtitle="Assign quantities to projects"
   869→                badge={`${projectFields.length} projects`}

   871→              <CollapsibleContent className="space-y-3 pt-3 px-1">
   872→                {projectFields.map((pf, idx) => {

   875→                  return (

   878→                      <div className="grid grid-cols-4 gap-2">

   915→          {/* ─── 5. Additional ──────────────────────────────────────────── */}
   916→          <Collapsible open={sectionsOpen.additional} onOpenChange={() => toggleSection('additional')}>
   917→            <SectionHeader sectionKey="additional" title="Additional" subtitle="Opening date, remarks, notes" />
   918→            <CollapsibleContent className="space-y-4 pt-3 px-1">
   919→              <div className="grid grid-cols-2 gap-4">
   920→                <div>
   921→                  <Label className="text-xs">Opening Date</Label>
   922→                  <Input type="date" value={openingDate} onChange={(e) => setOpeningDate(e.target.value)} />
   923→                </div>
   924→              </div>
   925→              <div>
   926→                <Label className="text-xs">Remarks</Label>
   927→                <Textarea value={remarks} onChange={(e) => setRemarks(e.target.value)} placeholder="Any remarks about this entry..." rows={2} />
   928→              </div>
   929→              <div>
   930→                <Label className="text-xs">Internal Notes</Label>
   931→                <Textarea value={internalNotes} onChange={(e) => setInternalNotes(e.target.value)} placeholder="Internal notes (not shown to everyone)..." rows={2} />
   932→              </div>
   933→            </CollapsibleContent>
   934→          </Collapsible>
   935→        </div>

   939→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   940→            Cancel
   941→          </Button>
   942→          <Button onClick={handleSubmit} disabled={submitting || (isEdit ? false : (!selectedProduct || openingQty <= 0))}>
   943→            {submitting && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   944→            {isEdit ? 'Update' : 'Create Opening Stock'}
   945→          </Button>
   946→        </DialogFooter>
   947→      </DialogContent>
   948→    </Dialog>
   949→  )
